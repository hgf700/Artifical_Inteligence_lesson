import torch
import torch.nn as nn
import torch.optim as optim
import torch.nn.functional as F

import torchvision.datasets as datasets
import torchvision.transforms as transforms
from torch.utils.data import DataLoader

from tqdm import tqdm
from torchmetrics.classification import Accuracy

# DATA
batch_size = 64

#zamiana na pixel 255 -> 1.0 pixel 0   -> 0.0
transform = transforms.ToTensor()

train_dataset = datasets.MNIST(root="dataset/", train=True, download=True, transform=transform)
test_dataset = datasets.MNIST(root="dataset/", train=False, download=True, transform=transform)

train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
test_loader = DataLoader(test_dataset, batch_size=batch_size, shuffle=False)

#KOD
class CNN(nn.Module):
    def __init__(self, in_channels=1, num_classes=10):
        super().__init__()

        # nakłada 8 masek 3x3 px i zwraca z [1, 28, 28] na [8, 28, 28] padding=1 zachowuje rozmiar
        self.conv1 = nn.Conv2d(in_channels, 8, kernel_size=3, padding=1)
        # Zmniejsza obraz 2x. 28x28 -> 14x14 -> 7x7
        self.pool = nn.MaxPool2d(2, 2)
        # 8 wejsc z wczesniej 16 wyjsciowych maska 3x3
        self.conv2 = nn.Conv2d(8, 16, kernel_size=3, padding=1)
        # wejsc 16 cech z conv2 7x7 po zmniejszeniu obrazu przez pool
        self.fc1 = nn.Linear(16 * 7 * 7, num_classes)

    # przeplyw
    def forward(self, x):
        # wykonanie conv1
        x = F.relu(self.conv1(x))
        # wykonanie pool 1x 14x14
        x = self.pool(x)
        # wykonanie conv2
        x = F.relu(self.conv2(x))
        # wykonanie pool 2x 7x7
        x = self.pool(x)
        # Flatten [64, 16, 7, 7] -> [64, 784] 64 jako rozmiar batch -1 = sam wyliczy wymiar.
        x = x.view(x.size(0), -1)
        # wykonanie conv1
        x = self.fc1(x)
        return x

# DEVICE
device = "cuda" if torch.cuda.is_available() else "cpu"

model = CNN().to(device)

# LOSS + OPT Liczy błąd klasyfikacji.
criterion = nn.CrossEntropyLoss()
# Adam aktualizuje wagi sieci.
optimizer = optim.Adam(model.parameters(), lr=0.001)

# TRAINING ilosc przejsc przez dane
num_epochs = 2

for epoch in range(num_epochs):
    print(f"Epoch [{epoch+1}/{num_epochs}]")
    model.train()

    for data, targets in tqdm(train_loader):
        data, targets = data.to(device), targets.to(device)

        # Model przewiduje wyniki.
        scores = model(data)
        # Obliczanie błędu.
        loss = criterion(scores, targets)

        # PyTorch sumuje gradienty, więc trzeba je resetować.
        optimizer.zero_grad()
        #Liczy gradienty.
        loss.backward()
        #Aktualizacja parametrów sieci.
        optimizer.step()

# TEST
accuracy = Accuracy(task="multiclass", num_classes=10).to(device)

#  Tryb ewaluacji. Wyłącza np. dropout/batchnorm training behavior.
model.eval()
# Nie liczy gradientów → szybciej i mniej pamięci.
with torch.no_grad():
    for images, labels in test_loader:
        images, labels = images.to(device), labels.to(device)

        outputs = model(images)
        # Przwiduje wyniki dla calego batch
        preds = torch.argmax(outputs, dim=1)

        #Zlicza poprawne odpowiedzi.
        accuracy.update(preds, labels)

# zlicza Poprawne / zle = %
print("Test accuracy:", accuracy.compute().item())
